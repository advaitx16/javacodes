    <br>
    <button onclick="greet()">Greet Me</button>
    <button onclick="changeColor()">Change Background</button>
    <p id="message"></p>
  </div>

  <!-- JavaScript -->
  <script>
    function greet() {
      var name = document.getElementById("nameInput").value;
      if (name === "") {
        document.getElementById("message").innerText = "Please enter your name!";
      } else {
        document.getElementById("message").innerText = "Hello, " + name + "! Welcome.";
      }
    }

    function changeColor() {
      var colors = ["#f0f4f8", "#ffeaa7", "#dfe6e9", "#fab1a0", "#a29bfe"];
      var random = Math.floor(Math.random() * colors.length);
      document.body.style.backgroundColor = colors[random];
    }
  </script>
